package it.yasde;

import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.lang.reflect.Method;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;

/**
 * Prove sul telefono, fuori dal jar di produzione. Si usa insieme al jar completo.
 *
 *   ... app_process / it.yasde.ProvaPannello servizio <fileChiamata> <fileInputExtra> <fileGuasto> <fileFrequenza> [limiteAssenzaMs]
 *     avvia il servizio vero con punti di sostituzione validi solo nella prova:
 *     - stato della chiamata finto (le chiamate vere non si possono fare): «in chiamata» finche' <fileChiamata> esiste;
 *     - percorsi di input in piu' che stanno in <fileInputExtra> (se esiste);
 *     - <fileGuasto>: se esiste, la prossima riaccensione fallisce (una volta) e il file si cancella;
 *     - <fileFrequenza>: se esiste, contiene la frequenza (Hz) che la misura riporta invece di quella vera.
 *
 *   ... app_process / it.yasde.ProvaPannello tastiera
 *     crea una tastiera virtuale (uhid, solo write: la shell puo') con un solo tasto, F20, che Android non
 *     usa per nulla. Scrive «creato», poi per ogni riga letta da stdin: «premi» = pressione e rilascio del
 *     tasto, «fine» = distrugge il dispositivo ed esce. I tocchi veri non si possono iniettare (la shell non
 *     puo' scrivere in /dev/input, e gli input iniettati da `input` non passano da li').
 */
public final class ProvaPannello {
    // struct uhid_event (packed): type u32 + unione; CREATE2 = 4376 byte, INPUT2 = type + u16 + 4096 byte
    private static final int UHID_DESTROY = 1;
    private static final int UHID_CREATE2 = 11;
    private static final int UHID_INPUT2 = 12;
    private static final int DIM_EVENTO = 4376;

    // tastiera HID standard (boot keyboard) senza LED: 1 byte modificatori, 1 riservato, 6 tasti. Il massimo
    // logico e' 0x7F (non 0x65): F20 vale 0x6F e fuori dall'intervallo il kernel scarta il tasto
    private static final int[] DESCRITTORE = {
        0x05, 0x01, 0x09, 0x06, 0xA1, 0x01, 0x05, 0x07, 0x19, 0xE0, 0x29, 0xE7, 0x15, 0x00, 0x25, 0x01,
        0x75, 0x01, 0x95, 0x08, 0x81, 0x02, 0x95, 0x01, 0x75, 0x08, 0x81, 0x01, 0x95, 0x06, 0x75, 0x08,
        0x15, 0x00, 0x25, 0x7F, 0x05, 0x07, 0x19, 0x00, 0x29, 0x7F, 0x81, 0x00, 0xC0,
    };
    private static final int USO_F20 = 0x6F;

    private ProvaPannello() {}

    public static void main(String[] args) throws Exception {
        if (args.length >= 1 && args[0].equals("tastiera")) {
            tastiera();
            return;
        }
        NascostePannello.chiamataFinta = new File(args[1]);
        InputFisico.fileExtra = new File(args[2]);
        NascostePannello.accensioneFalliscePrima = new File(args[3]);
        NascostePannello.frequenzaFinta = new File(args[4]);
        // solo per le prove: limite dell'assenza del PC accorciato (ms); puo' solo accorciare, vale prima di registra()
        if (args.length >= 6) Pannello.limitePerProve(Long.parseLong(args[5]));
        Method m = Class.forName("it.yasde.Servizio").getMethod("main", String[].class);
        m.invoke(null, (Object) new String[0]);
    }

    private static ByteBuffer evento(int tipo, int dim) {
        ByteBuffer b = ByteBuffer.allocate(dim).order(ByteOrder.LITTLE_ENDIAN);
        b.putInt(tipo);
        return b;
    }

    private static void tastiera() throws Exception {
        FileOutputStream uhid = new FileOutputStream("/dev/uhid");
        try {
            ByteBuffer c = evento(UHID_CREATE2, DIM_EVENTO);
            byte[] nome = "yasde-prova".getBytes("UTF-8");
            c.put(nome);
            c.position(4 + 128 + 64 + 64); // dopo name, phys, uniq
            c.putShort((short) DESCRITTORE.length);
            c.putShort((short) 0x03); // BUS_USB
            c.putInt(0x1d6b); // vendor: valore finto
            c.putInt(0x0104); // product: valore finto
            c.putInt(0); // version
            c.putInt(0); // country
            for (int v : DESCRITTORE) c.put((byte) v);
            uhid.write(c.array());
            System.out.println("creato");
            System.out.flush();
            InputStream in = System.in;
            StringBuilder riga = new StringBuilder();
            int ch;
            while ((ch = in.read()) >= 0) {
                if (ch != '\n') {
                    riga.append((char) ch);
                    continue;
                }
                String r = riga.toString();
                riga.setLength(0);
                if (r.equals("premi")) {
                    rapporto(uhid, USO_F20);
                    Thread.sleep(50);
                    rapporto(uhid, 0);
                    System.out.println("premuto");
                    System.out.flush();
                } else if (r.equals("fine")) {
                    break;
                }
            }
        } finally {
            try {
                uhid.write(evento(UHID_DESTROY, DIM_EVENTO).array());
            } finally {
                uhid.close();
            }
        }
    }

    private static void rapporto(FileOutputStream uhid, int uso) throws Exception {
        ByteBuffer b = evento(UHID_INPUT2, DIM_EVENTO);
        b.putShort((short) 8);
        b.put(new byte[] {0, 0, (byte) uso, 0, 0, 0, 0, 0});
        uhid.write(b.array());
    }
}
