package it.yasde;

import java.io.ByteArrayInputStream;
import java.io.DataInputStream;
import java.util.ArrayList;
import java.util.List;

/** Prova su JVM della parte pura dei dati del telefono (nessun Android). */
public final class ProvaDati {
    private static int falliti;

    private static void controlla(boolean ok, String nome) {
        System.out.println((ok ? "ok      " : "FALLITA ") + nome);
        if (!ok) falliti++;
    }

    private static String ripeti(String s, int n) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) sb.append(s);
        return sb.toString();
    }

    private static String registro(String pk, String chiave, String gruppo, String flags, String titolo, String testo, String nascosta, long tempo) {
        String extras = "            extras={\n                android.title=" + titolo + "\n"
                + (testo == null ? "" : "                android.text=" + testo + "\n")
                + "                android.showWhen=Boolean (true)\n            }\n";
        return "    NotificationRecord(0x1: pkg=" + pk + " user=UserHandle{0} id=1 tag=null importance=3 key=" + chiave + ": Notification(channel=c)\n"
                + "      uid=1 userId=0\n      opPkg=" + pk + "\n      flags=" + flags + "\n      pri=0\n      key=" + chiave + "\n"
                + "      seen=false\n      groupKey=" + gruppo + "\n      notification=\n            fullscreenIntent=null\n" + extras
                + "      publicNotification=\n            extras={\n                android.title=String (PUBBLICA)\n            }\n"
                + "      stats=x\n      mHidden==" + nascosta + "\n      mCreationTimeMs=" + tempo + "(2026-01-01 00:00:00.000+0100)\n      mUpdateTimeMs=1(x)\n";
    }

    /** Record con il blocco extras scritto a mano (righe gia' con il loro rientro) e un `when` facoltativo. */
    private static String registroGrezzo(String pk, String chiave, String extras, String when, boolean conNotifica) {
        return "    NotificationRecord(0x1: pkg=" + pk + " user=UserHandle{0} id=1 tag=null key=" + chiave + ": Notification(channel=c)\n"
                + "      uid=1 userId=0\n      flags=0\n      key=" + chiave + "\n      seen=false\n      groupKey=" + chiave + "\n"
                + (conNotifica ? "      notification=\n            number=0\n" + (when == null ? "" : "            when=" + when + "/" + when + "\n") + extras : "")
                + "      publicNotification=\n            None\n      stats=x\n      mHidden==false\n      mCreationTimeMs=777(x)\n";
    }

    private static String extras(String voci) {
        return "            extras={\n" + voci + "            }\n";
    }

    private static boolean ok(long da, long a, int massimo) {
        try {
            CampiDati.controllaIntervallo(da, a, massimo);
            return true;
        } catch (IllegalArgumentException e) {
            return false;
        }
    }

        String coda = "isAppCastingOn:false\nUser Records:\nRecord for full_user=0\n";
        String dumpMedia = intest + sess1 + sess2 + coda;
        CampiTelefonia.Media dmz = LettoriShell.mediaDaDumpsys(dumpMedia);
        controlla(dmz != null && dmz.esito == 0 && dmz.stato == 2 && dmz.pacchetto.equals("org.sample.player") && dmz.titolo.equals("Sample Song")
                && dmz.artista.equals("Sample Artist") && dmz.comandi == 7, "dumpsys media: sceglie la sessione in riproduzione (PLAYING(3)), titolo, artista, comandi dalle azioni");
        CampiTelefonia.Media dmA = LettoriShell.mediaDaDumpsys(intest + sess2.replace("PLAYING(3)", "PAUSED(2)") + sess1.replace("PAUSED(2)", "NONE(0)") + coda);
        controlla(dmA != null && dmA.pacchetto.equals("org.sample.player") && dmA.stato == 1 && dmA.comandi == 6, "dumpsys media: una sessione in pausa; play/pausa solo con PLAY o PLAY_PAUSE (qui no)");
        CampiTelefonia.Media dmB = LettoriShell.mediaDaDumpsys(intest + sess1 + coda);
        controlla(dmB != null && dmB.stato == 1 && dmB.pacchetto.equals("org.finto.prima") && dmB.titolo.equals("Altro Brano") && dmB.comandi == 3,
                "dumpsys media: in pausa con PLAY e avanti = comandi 1 e 2");
        CampiTelefonia.Media dmC = LettoriShell.mediaDaDumpsys(dumpMedia.replace("actions=" + (2 | 32 | 16), "actions=x"));
        controlla(dmC != null && dmC.stato == 2 && dmC.comandi == 0, "dumpsys media: azioni illeggibili = nessun comando (0)");
        CampiTelefonia.Media dmD = LettoriShell.mediaDaDumpsys(dumpMedia.replace("description=Sample Song, Sample Artist, Sample Album", "description=null, null, null"));
        controlla(dmD != null && dmD.titolo.isEmpty() && dmD.artista.isEmpty(), "dumpsys media: descrizione null = vuota");
        CampiTelefonia.Media dmE = LettoriShell.mediaDaDumpsys(dumpMedia.replace("PLAYING(3)", "3"));
        controlla(dmE != null && dmE.stato == 2 && dmE.titolo.equals("Sample Song"), "dumpsys media: anche il solo numero dello stato");
        // il caso reale: stato NONE(0), azioni 8192, ma il brano e' presente -> in pausa (1), niente comandi
        String reale = intest.replace("Total sessions: 2 across 2", "Total sessions: 1 across 1").replace("have 2 sessions", "have 1 sessions")
                + sess2.replace("PLAYING(3)", "NONE(0)").replace("actions=" + (2 | 32 | 16), "actions=8192") + coda;
        CampiTelefonia.Media dmR = LettoriShell.mediaDaDumpsys(reale);
        controlla(dmR != null && dmR.esito == 0 && dmR.stato == 1 && dmR.titolo.equals("Sample Song") && dmR.artista.equals("Sample Artist")
                && dmR.pacchetto.equals("org.sample.player") && dmR.comandi == 0, "dumpsys media: caso reale, stato NONE(0) con brano = in pausa, comandi 0");
        String senzaBrano = reale.replace("description=Sample Song, Sample Artist, Sample Album", "description=null, null, null");
        CampiTelefonia.Media dmS = LettoriShell.mediaDaDumpsys(senzaBrano);
        controlla(dmS != null && dmS.esito == 0 && dmS.stato == 0 && dmS.titolo.isEmpty() && dmS.pacchetto.isEmpty() && dmS.comandi == 0,
                "dumpsys media: stato NONE(0) senza brano = nessuna sessione (stato 0, esito 0), non «non disponibile»");
        controlla(LettoriShell.mediaDaDumpsys(senzaBrano.replace("metadata: size=7, description=null, null, null", "metadata: null")).stato == 0
                && LettoriShell.mediaDaDumpsys(senzaBrano.replace("NONE(0)", "STOPPED(1)")).stato == 0 && LettoriShell.mediaDaDumpsys(senzaBrano.replace("NONE(0)", "ERROR(7)")).stato == 0,
                "dumpsys media: metadata null, STOPPED, ERROR senza brano = nessuna sessione");
        CampiTelefonia.Media dmF = LettoriShell.mediaDaDumpsys(reale.replace("NONE(0)", "BUFFERING(6)"));
        controlla(dmF != null && dmF.stato == 2, "dumpsys media: BUFFERING conta come in riproduzione");
        // descrizione divisa da destra
        String[][] desc = {
            {"Titolo, Artista, Album", "Titolo", "Artista"},
            {"Ti, to, Artista, Album", "Ti, to", "Artista"},
            {"Solo titolo", "Solo titolo", ""},
            {"Titolo, Artista", "Titolo", "Artista"},
            {"null, null, null", "", ""},
            {"Titolo, null, Album", "Titolo", ""},
            {"A, b, C, d, Album", "A, b, C", "d"}};
        boolean descOk = true;
        for (String[] dd : desc) {
            CampiTelefonia.Media mm = LettoriShell.mediaDaDumpsys(reale.replace("description=Sample Song, Sample Artist, Sample Album", "description=" + dd[0]));
            boolean vuota = dd[1].isEmpty() && dd[2].isEmpty();
            descOk &= mm != null && mm.titolo.equals(dd[1]) && mm.artista.equals(dd[2]) && (vuota ? mm.stato == 0 : mm.stato == 1);
        }
        controlla(descOk, "dumpsys media: descrizione divisa da destra (2 virgole, 3 virgole, nessuna virgola, due campi, null)");
        String zero = "MEDIA SESSION SERVICE\n=== PACKAGE SUMMARY ===\nTotal sessions: 0 across 0 package(s)\n=== DETAILED SESSION INFO ===\nGlobal priority session is x/1 (userId=0)\n  Sessions Stack - have 0 sessions:\nUser Records:\n";
        CampiTelefonia.Media dvz = LettoriShell.mediaDaDumpsys(zero);
        controlla(dvz != null && dvz.stato == 0 && dvz.esito == 0 && dvz.titolo.isEmpty(), "dumpsys media: zero sessioni = stato 0");
        controlla(LettoriShell.mediaDaDumpsys("") == null && LettoriShell.mediaDaDumpsys("Can't find service: media_session") == null && LettoriShell.mediaDaDumpsys(null) == null
                && LettoriShell.mediaDaDumpsys(reale.replace("state=PlaybackState", "state=Boh")) == null,
                "dumpsys media: testo non capito (sessioni dichiarate ma nessuna leggibile) = null (esito 1), mai dati a caso");
        controlla(LettoriShell.nomeDaWifiStatus("Wifi is enabled\nWifi is connected to \"Rete Finta\"\nWifiInfo: SSID: \"Altra\", BSSID: aa").equals("Rete Finta")
                && LettoriShell.nomeDaWifiStatus("x\nWifiInfo: SSID: \"Solo, Virgola\", BSSID: aa:bb, MAC: cc").equals("Solo, Virgola")
                && LettoriShell.nomeDaWifiStatus("Wifi is connected to <unknown ssid>").isEmpty() && LettoriShell.nomeDaWifiStatus("Wifi is disabled").isEmpty()
                && LettoriShell.nomeDaWifiStatus(null).isEmpty() && LettoriShell.nomeDaWifiStatus("WifiInfo: SSID: <unknown ssid>, BSSID: <none>").isEmpty(), "cmd wifi status: nome della rete");
        String regT = "  Phone Id=0\n    mServiceState={mVoiceRegState=1(OUT_OF_SERVICE), mDataRegState=1(OUT_OF_SERVICE), mOperatorAlphaLong=, mOperatorAlphaShort=, getRilDataRadioTechnology=0(Unknown), getRilVoiceRadioTechnology=0(Unknown), x=y}\n"
                + "    mSignalStrength=SignalStrength:{mLte=Invalid}\n"
                + "  Phone Id=1\n    mServiceState={mVoiceRegState=0(IN_SERVICE), mDataRegState=0(IN_SERVICE), mOperatorAlphaLong=Op Finto, mOperatorAlphaShort=OF, getRilVoiceRadioTechnology=14(LTE), getRilDataRadioTechnology=20(NR), x=y}\n"
                + "    mSignalStrength=SignalStrength:{mCdma=Invalid,mLte=CellSignalStrengthLte: rssi=-81 level=2 parametersUseForLevel=0,mNr=CellSignalStrengthNr: level=3,SignalBarInfo{ lteLevel=1 }}\n";
        LettoriShell.Mobile moA = LettoriShell.mobileDaRegistry(regT);
        controlla(moA != null && moA.mobile == 2 && moA.livello == 3 && moA.generazione == 5 && moA.operatore.equals("Op Finto"), "registro telefonia: primo slot in servizio, livello, generazione, operatore");
        LettoriShell.Mobile moB = LettoriShell.mobileDaRegistry(regT.replace("getRilDataRadioTechnology=20(NR)", "getRilDataRadioTechnology=0(Unknown)").replace("mOperatorAlphaLong=Op Finto", "mOperatorAlphaLong="));
        controlla(moB != null && moB.generazione == 4 && moB.operatore.equals("OF"), "registro telefonia: generazione dalla voce e operatore breve se manca il lungo");
        LettoriShell.Mobile moC = LettoriShell.mobileDaRegistry(regT.replace("0(IN_SERVICE)", "1(OUT_OF_SERVICE)"));
        controlla(moC != null && moC.mobile == 1 && moC.livello == 0 && moC.operatore.isEmpty(), "registro telefonia: nessuno slot in servizio = senza servizio");
        controlla(LettoriShell.nomeDaWifiStatus("  BSSID: aa\nWifiInfo: SSID: \"Reale\", BSSID: aa:bb, MAC: cc").equals("Reale")
                && LettoriShell.nomeDaWifiStatus("    BSSID: \"Falso\"\n").isEmpty()
                && LettoriShell.nomeDaWifiStatus("Wifi is connected to \"Troncat").isEmpty()
                && LettoriShell.nomeDaWifiStatus("WifiInfo: SSID: \"Troncat").isEmpty() && LettoriShell.nomeDaWifiStatus("Wifi is connected to \"").isEmpty(),
                "cmd wifi status: BSSID non e' SSID, virgoletta senza chiusura = vuoto");
        controlla(LettoriShell.mobileDaRegistry("niente di utile") == null && LettoriShell.mobileDaRegistry(null) == null, "registro telefonia: testo non capito = null");
        controlla(LettoriShell.generazioneDaNome("LTE") == 4 && LettoriShell.generazioneDaNome("HSPAP") == 3 && LettoriShell.generazioneDaNome("EDGE") == 2
                && LettoriShell.generazioneDaNome("Unknown") == 0 && LettoriShell.generazioneDaNome(null) == 0, "generazione dal nome della tecnologia");

        System.out.println(falliti == 0 ? "tutte le prove superate" : falliti + " prove fallite");
        if (falliti != 0) System.exit(1);
    }
